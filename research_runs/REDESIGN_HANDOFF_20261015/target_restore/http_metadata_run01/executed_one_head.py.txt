"""Single authorized frozen-URL HEAD, no body and no redirect/retry."""
import datetime, hashlib, json, ssl, urllib.error, urllib.request
from pathlib import Path
from urllib.parse import urlsplit

BASE=Path('/workspace/mi-eeg-redesign-qa-20261012/target-http-head-run01')
OUT=BASE/'head_observation.json'
if OUT.exists():
    raise SystemExit('already_observed_no_retry')
manifest_path=Path('/workspace/mi-eeg-redesign-20261012/research_runs/REDESIGN_SOURCE_SELECTION_20261012/evidence/dataset_sources/dreyer_nemar_acquisition_edf_manifest.json')
b=manifest_path.read_bytes()
assert hashlib.sha256(b).hexdigest()=='2cfdf35aa7990df880826d1b84d1230e0bda3c9393c17d3ec6e27b7b311d2bf2'
m=json.loads(b)
f=sorted(m['files'],key=lambda x:x['path'])[0]
url=f['bytes_url']
assert url=='https://data.nemar.org/nm000250/v1.0.5/sourcedata/sub-01/eeg/sub-01_task-R1acquisition_eeg.edf'
class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        raise urllib.error.HTTPError(req.full_url,code,'redirect_not_followed',headers,fp)
context=ssl.create_default_context()
assert context.verify_mode==ssl.CERT_REQUIRED and context.check_hostname
opener=urllib.request.build_opener(urllib.request.HTTPSHandler(context=context),NoRedirect())
r={'schema_version':1,'request_url':url,'request_method':'HEAD','request_headers':{'Accept-Encoding':'identity'},'attempts':1,'redirects_followed':0,'response_body_read_calls':0,'edf_body_bytes_read':0,'tls_certificate_verified_by_client':True,'tls_hostname_check_enabled':True,'default_proxy_and_CA_preserved':True,'frozen_directory':{'dataset':m['dataset'],'version':m['version'],'tree_sha':m['tree_sha']},'frozen_first_object_bytes':f['size'],'frozen_first_object_sha256':f['checksum'],'prior_GET_HTTP_status':'NOT_RECORDED_NOT_INFERRED','observation_scope':'current Codex executor, not live Pod','training_authorization':False,'started_at_utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
response=None
try:
    try:
        response=opener.open(urllib.request.Request(url,headers={'Accept-Encoding':'identity'},method='HEAD'),timeout=30)
    except urllib.error.HTTPError as exc:
        response=exc
    r['http_status']=response.code
    r['status']='HTTP_METADATA_OBSERVED_NO_BODY_NO_REDIRECT'
    r['response_url']=response.geturl()
    allow={'date','server','content-type','content-length','content-encoding','etag','last-modified','via','x-cache','x-cache-hits'}
    r['response_headers']={k.lower():v for k,v in response.headers.items() if k.lower() in allow}
    location=response.headers.get('Location')
    r['location_present']=location is not None
    if location is not None:
        r['raw_location_sha256']=hashlib.sha256(location.encode('utf-8')).hexdigest()
        p=urlsplit(location)
        public=all(ord(c)>=32 and ord(c)!=127 for c in location) and not(p.username or p.password or p.query or p.fragment)
        r['location_public_no_credentials_query_fragment']=public
        r['controlled_public_location']=location if public else None
        r['location_redacted_host']=p.hostname
        r['location_redacted_path']=p.path
    r['explicit_auth_denial']=response.code in (401,403)
except Exception as exc:
    r['status']='HEAD_FAILED_NO_RETRY'
    r['error_class']=type(exc).__name__
finally:
    if response is not None:
        response.close()
    r['finished_at_utc']=datetime.datetime.now(datetime.timezone.utc).isoformat()
    with OUT.open('x') as stream:
        json.dump(r,stream,indent=2,sort_keys=True);stream.write('\n')
print(json.dumps(r,indent=2,sort_keys=True))
